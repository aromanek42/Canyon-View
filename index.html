document.addEventListener("DOMContentLoaded", () => { // This specific endpoint returns pure JSON instead of text, which bypasses the GitHub Pages CORS block const endpoint = "https://weather.gov"; fetch(endpoint, { headers: { // NOAA requires a custom User-Agent string to prevent traffic throttling "User-Agent": "GitHubPagesMarineApp (your-email@example.com)" } }) .then(response => { if (!response.ok) { throw new Error(`NOAA API returned status: ${response.status}`); } return response.json(); }) .then(data => { const container = document.getElementById("forecast-container"); const timeElement = document.getElementById("update-time"); // Update the timestamp header if (data.properties && data.properties.updated) { const updateDate = new Date(data.properties.updated); timeElement.textContent = `Updated: ${updateDate.toLocaleString()}`; } // Loop through the split up forecast periods (Today, Tonight, Tomorrow, etc.) const periods = data.properties.periods; if (periods && periods.length > 0) { container.innerHTML = ""; // Clear out the loading text periods.forEach(period => { const periodDiv = document.createElement("div"); periodDiv.className = "period"; const nameDiv = document.createElement("div"); nameDiv.className = "period-name"; nameDiv.textContent = period.name; const textDiv = document.createElement("div"); textDiv.className = "period-text"; textDiv.textContent = period.detailedForecast; periodDiv.appendChild(nameDiv); periodDiv.appendChild(textDiv); container.appendChild(periodDiv); }); } else { container.innerHTML = `
No periods returned in the marine forecast JSON.
`; } }) .catch(error => { console.error("Fetch Error:", error); document.getElementById("forecast-container").innerHTML = `
Unable to display live data.
${error.message}
`; }); });
