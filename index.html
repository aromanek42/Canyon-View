// FIXED: Changed from 'https://weather.gov' to the exact API endpoint URL const apiZoneUrl = 'https://weather.gov'; async function fetchMarineWeather() { try { const response = await fetch(apiZoneUrl, { method: 'GET', headers: { // NOAA requires a unique User-Agent identifier to grant access to the API 'User-Agent': 'MyHudsonCanyonDashboard (contact: yourname@example.com)' } }); if (!response.ok) { throw new Error(`NWS API error: ${response.status}`); } const data = await response.json(); // Extract properties from the JSON-LD response const periods = data.properties.periods; const updatedTime = new Date(data.properties.updated).toLocaleString(); // Format the structured text nicely let cleanForecastHTML = ''; periods.forEach(period => { cleanForecastHTML += `${period.name}:
${period.detailedForecast}

`; }); // Update your HTML elements document.getElementById('forecast-data').innerHTML = cleanForecastHTML; document.getElementById('update-time').textContent = `NOAA Data Updated: ${updatedTime}`; } catch (error) { console.error('Error fetching data:', error); document.getElementById('forecast-data').innerHTML = `Failed to load live conditions.
` + `Reason: ${error.message}

` + `Try viewing the data directly via the NWS Zone Page.`; } } // Initial fetch on browser load fetchMarineWeather(); // Automatically update every 30 minutes setInterval(fetchMarineWeather, 1800000);
